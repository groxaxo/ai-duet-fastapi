│ USAGE EXAMPLES                                                         │
└───────────────────────────────────────────────────────────────────────┘

Update existing agent avatar:
  ws.send(JSON.stringify({
    type: "set_agent",
    agent: "A",
    avatar: "🧙"
  }));

Add new agent with avatar:
  ws.send(JSON.stringify({
    type: "add_agent",
    agent_id: "C",
    name: "Charlie",
    avatar: "🎭",
    instructions: "You are a dramatic storyteller",
    voice: "am_adam",
    lang: "en"
  }));
